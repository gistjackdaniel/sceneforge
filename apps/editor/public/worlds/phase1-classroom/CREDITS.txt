    npm install --prefix .local/phase1-world --no-save @gltf-transform/core@4.5.1 @gltf-transform/extensions@4.5.1 @gltf-transform/functions@4.5.1 meshoptimizer@1.0.1 ndarray-pixels@5.2.0 sharp@0.35.5
    node apps/editor/scripts/prepare-phase1-world.mjs
